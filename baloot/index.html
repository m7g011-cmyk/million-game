// شراء الورقة والتوزيع الصحيح لـ 8 أوراق لكل لاعب function makeBid(choice) { clearInterval(timerInterval); db.ref('baloot_rooms/' + currentRoomCode).get().then(snap => { let data = snap.val(); if (choice === 'pass') { let pCount = (data.passesCount || 0) + 1; let nextTurn = (data.turnIndex + 1) % 4; if (pCount === 4 && data.bidRound === 1) { data.bidRound = 2; data.passesCount = 0; data.turnIndex = (data.dealerIndex + 1) % 4; } else if (pCount === 4 && data.bidRound === 2) { let nextDealer = (data.dealerIndex + 1) % 4; startNewRound(data.players, nextDealer); return; } else { data.passesCount = pCount; data.turnIndex = nextTurn; } } else { let buyerIndex = data.turnIndex; let buyer = data.players[buyerIndex]; // 1. المشتري يأخذ ورقة الوسط المكشوفة buyer.cards.push(data.revealedCard); data.revealedCard = null; // 2. التوزيع التكملي لضبط اليد على 8 أوراق بالضبط للجميع data.players.forEach((p, idx) => { if (idx === buyerIndex) { // المشتري (معه 5 + 1 المكشوفة = 6): يُعطى ورقتين فقط لتصبح 8 p.cards.push(data.deck.pop(), data.deck.pop()); } else { // باقي اللاعبين (معهم 5): يُعطى كل واحد 3 أوراق لتصبح 8 p.cards.push(data.deck.pop(), data.deck.pop(), data.deck.pop()); } }); data.mode = 'playing_cards'; data.contractType = choice; data.turnIndex = (data.dealerIndex + 1) % 4; // يبدأ اللعب فوراً من يمين الموزع } db.ref('baloot_rooms/' + currentRoomCode).set(data); }); }
